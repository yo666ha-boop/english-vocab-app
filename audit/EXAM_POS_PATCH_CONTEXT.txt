function uniqueKeepOrder(values) {
  const out = [];
  const seen = new Set();
  values.forEach(v => {

}

function shuffle(arr) {
  const a = arr.slice();
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

      (!state.toggles.starOnly || String(r.star) === '1') &&

    matches(state.filters.elemType, r.elem_type) &&
    matches(state.filters.elemCategory, r.elem_category) &&

  );
}

function shuffle(arr) {
  const a = arr.slice();
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}
function createTest() {
  state.currentList = filteredRows();
  renderSummary();
  renderList();
  const rows = state.currentList.slice();
  if (rows.length === 0) {
    state.currentTest = [];
    els.testSummary.textContent = '該当語がありません';
    els.testArea.innerHTML = '';
    return;
  }
  const specialRule = els.starQuizRule && els.starQuizRule.value === 'star_both_nonstar_read';
  const candidates = [];
  if (specialRule) {
    rows.forEach(r => {
      if (String(r.star || '') === '1') {
        candidates.push({ row: r, fixedMode: 'en2jp' });
        candidates.push({ row: r, fixedMode: 'jp2en' });
      } else {
        candidates.push({ row: r, fixedMode: 'en2jp' });
      }
    });
  } else {
    rows.forEach(r => candidates.push({ row: r, fixedMode: '' }));
  }
  const count = Math.min(parseInt(els.qCount.value, 10), candidates.length);
  const picked = shuffle(candidates).slice(0, count).map((item, idx) => {
    const r = item.row;
    let mode = item.fixedMode || els.quizMode.value;
    if (mode === 'random') mode = Math.random() < 0.5 ? 'en2jp' : 'jp2en';
    const prompt = mode === 'en2jp' ? r.english : r.japanese;
    const answer = mode === 'en2jp' ? r.japanese : r.english;
    return { no: idx + 1, mode, prompt, answer, kana: '', extra: '', star: String(r.star || '') === '1' };
  });
  state.currentTest = picked;
  state.answersVisible = false;
  renderTest();
}

function esc(s) {
  return String(s ?? '').replace(/[&<>"]/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]));
}
function uniqueKeepOrder(values) {
  const out = [];
  const seen = new Set();
  values.forEach(v => {
    const sv = String(v ?? '');
    if (sv && !seen.has(sv)) { seen.add(sv); out.push(sv); }
  });
  return out;
}
function rowsForDataset() {
  return DATA.filter(r => r.dataset === state.dataset);
}
function selectedArray(key) {
  return Array.from(state.filters[key] || []);
}
function matches(setObj, value) {
  if (!setObj || setObj.size === 0) return true;
  return setObj.has(String(value ?? ''));
}
function renderChecklist(box, key, values) {
  const selected = state.filters[key];
  box.innerHTML = values.map(v => {
    const sv = String(v);
    return `<label class="checkitem"><input type="checkbox" data-filter="${key}" value="${esc(sv)}" ${selected.has(sv) ? 'checked' : ''}> <span>${esc(sv)}</span></label>`;
  }).join('');
  box.querySelectorAll('input[type="checkbox"]').forEach(input => {
    input.addEventListener('change', e => {
      const k = e.target.dataset.filter;
      const val = e.target.value;
      if (e.target.checked) state.filters[k].add(val);
      else state.filters[k].delete(val);
      clearCurrentTest();
      if (state.dataset === 'textbook' && ['textbook','grade','major','section'].includes(k)) renderAllFilters();
      if (state.dataset === 'exam' && ['examCategory'].includes(k)) renderAllFilters();
      if (state.dataset === 'elementary' && ['elemType'].includes(k)) renderAllFilters();
      refresh();
    });
  });
  updateMeta(key);
}

function renderAllFilters() {
  const rows = rowsForDataset();

function shuffle(arr) {
  const a = arr.slice();
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

    let mode = item.fixedMode || els.quizMode.value;
    if (mode === 'random') mode = Math.random() < 0.5 ? 'en2jp' : 'jp2en';
    const prompt = mode === 'en2jp' ? r.english : r.japanese;
    const answer = mode === 'en2jp' ? r.japanese : r.english;
    return { no: idx + 1, mode, prompt, answer, kana: '', extra: '', star: String(r.star || '') === '1' };
  });
  state.currentTest = picked;
  state.answersVisible = false;
  renderTest();
}

function compactList(arr, emptyLabel='すべて') {
  const vals = Array.from(arr || []).filter(v => String(v || '').trim());
  if (!vals.length) return emptyLabel;
  if (vals.length <= 6) return vals.join('、');
  return vals.slice(0, 6).join('、') + `、ほか${vals.length - 6}件`;
}

    rows.push(['大単元', compactList(state.filters.major, 'すべて')]);
    rows.push(['小単元', compactList(state.filters.section, 'すべて')]);
    rows.push(['品詞', compactList(state.filters.pos, 'すべて')]);
    const cond = [];
    if (state.toggles.starOnly) cond.push('☆のみ');
    if (state.toggles.phraseOnly) cond.push('熟語・表現のみ');
    if (state.toggles.currentOnly) cond.push('現行確認済みのみ');
    rows.push(['条件', cond.length ? cond.join('、') : '指定なし']);
  } else if (state.dataset === 'exam') {
    rows.push(['入試カテゴリ', compactList(state.filters.examCategory, 'すべて')]);
    rows.push(['分類', compactList(state.filters.examSub, 'すべて')]);
  } else {
    rows.push(['テスト区分', compactList(state.filters.elemType, 'すべて')]);
    rows.push(['カテゴリー', compactList(state.filters.elemCategory, 'すべて')]);
  }
  rows.push(['対象語数', `${state.currentList.length}語`]);
  rows.push(['出題形式', quizModeTextForPrint()]);

}

}

}
