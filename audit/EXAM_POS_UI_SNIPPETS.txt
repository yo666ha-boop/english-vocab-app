function shuffle(arr) {
  const a = arr.slice();
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

function compactList(arr, emptyLabel='すべて') {
  const vals = Array.from(arr || []).filter(v => String(v || '').trim());
  if (!vals.length) return emptyLabel;

}

function shortItemsForPrint(items, emptyLabel='全範囲') {

function shuffle(arr) {
  const a = arr.slice();
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}
